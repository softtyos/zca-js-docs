export type CreatePollOptions = {
    /**
     * Question for the poll.
     */
    question: string;

    /**
     * List of options for the poll.
     */
    options: string[];

    /**
     * Poll expiration time in milliseconds (0 = no expiration).
     */
    expiredTime?: number;

     * Allows multiple choices in the poll.
     */

};
